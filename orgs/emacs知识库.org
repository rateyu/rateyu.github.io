| 快捷键       | 功能                                     |

* 三、窗口与缓冲区管理
** 窗口切换
#+begin_src emacs-lisp
;; 方式1：windmove（推荐）
(global-set-key (kbd "C-<left>")  'windmove-left)
(global-set-key (kbd "C-<right>") 'windmove-right)
(global-set-key (kbd "C-<up>")    'windmove-up)

** 代码跳转（etags + xref）
#+begin_src shell
# 生成 TAGS 文件（多语言版本）
find . \( -name "*.java" -o -name "*.js" -o -name "*.py" \
  -o -name "*.go" -o -name "*.cpp" -o -name "*.hpp" \
  -o -name "*.sh" -o -name "*.yaml" -o -name "*.yml" \) \

(global-set-key (kbd "C-c f") 'xref-find-references)
#+end_src

** org-capture 配置（快速记录）
#+begin_src emacs-lisp
(setq org-capture-templates
      '(("t" "Task" entry
         (file+headline "~/orgs/tasks.org" "Tasks")
         "* TODO %?\n  %U")
        ("n" "Note" entry
         (file+headline "~/orgs/notes.org" "Notes")
         "* %? :NOTE:\n  %U")
        ("i" "Idea" entry
         (file+headline "~/orgs/ideas.org" "Ideas")
         "* %U %?\n  %i\n  %a")
        ("j" "Journal" entry
         (file+datetree "~/orgs/journal.org")
         "* %U %?\n  %i\n  %a")))

(global-set-key (kbd "C-c t") (lambda () (interactive) (org-capture nil "t")))
(global-set-key (kbd "C-c n") (lambda () (interactive) (org-capture nil "n")))
(global-set-key (kbd "C-c i") (lambda () (interactive) (org-capture nil "i")))
(global-set-key (kbd "C-c j") (lambda () (interactive) (org-capture nil "j")))
#+end_src

* 十二、键位配置
** Caps Lock ↔ Ctrl 交换（Linux）
#+begin_src shell
# ~/.keymaprc
remove Control = Control_R
remove mod1 = Alt_R
keysym Control_R = Alt_R
keysym Alt_R = Control_R
add Control = Control_R
add mod1 = Alt_R

# 执行
xmodmap ~/.keymaprc
#+end_src

** 写入启动脚本
#+begin_src shell
touch swkey && echo "xmodmap ~/.keymaprc" > swkey
chmod +x swkey && sudo mv swkey /usr/bin
#+end_src
